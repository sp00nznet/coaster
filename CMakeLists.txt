cmake_minimum_required(VERSION 3.20)
project(coaster-recomp LANGUAGES C CXX)

set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)

if(MSVC)
    add_compile_options(/utf-8)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
if(MINGW OR CMAKE_C_COMPILER_ID STREQUAL "GNU")
    add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

# ─── SDL2 (from vcpkg or system) ───
find_package(SDL2 CONFIG REQUIRED)

include_directories(${CMAKE_SOURCE_DIR}/include)
include_directories(${CMAKE_SOURCE_DIR}/RecompiledFuncs)

# ─── HAL + DOS compatibility ───
add_library(coaster_hal STATIC
    src/hal/video.c
    src/hal/input.c
    src/hal/timer.c
    src/recomp/cpu.c
    src/recomp/dos_compat.c
)
target_include_directories(coaster_hal PUBLIC
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/RecompiledFuncs
)

# ─── SDL2 platform ───
add_library(coaster_platform STATIC src/platform/sdl_platform.c)
target_include_directories(coaster_platform PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_link_libraries(coaster_platform PUBLIC SDL2::SDL2 SDL2::SDL2main)

# ─── Recompiled game code (auto-generated by tools/recomp.py) ───
# CONFIGURE_DEPENDS so re-running the recompiler (which may add/remove sharded
# files) is picked up without a manual reconfigure.
file(GLOB RECOMP_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_SOURCE_DIR}/RecompiledFuncs/coaster_recomp_*.c"
    "${CMAKE_SOURCE_DIR}/RecompiledFuncs/coaster_stubs.c"
    "${CMAKE_SOURCE_DIR}/RecompiledFuncs/coaster_dispatch.c"
    "${CMAKE_SOURCE_DIR}/RecompiledFuncs/coaster_impl.c"
)

if(RECOMP_SOURCES)
    add_library(coaster_recomp STATIC ${RECOMP_SOURCES})
    target_include_directories(coaster_recomp PUBLIC
        ${CMAKE_SOURCE_DIR}/include
        ${CMAKE_SOURCE_DIR}/RecompiledFuncs
    )
    target_link_libraries(coaster_recomp PUBLIC coaster_hal)

    # Auto-generated code: quiet the expected noise, and DON'T optimize it - the
    # giant straight-line CPU-emulation functions make the optimizer take minutes
    # for no runtime benefit.
    if(MSVC)
        target_compile_options(coaster_recomp PRIVATE
            /Od /W0 /wd4101 /wd4102 /wd4244 /wd4267 /wd4146)
    else()
        target_compile_options(coaster_recomp PRIVATE
            -O0 -w)
    endif()

    add_executable(coaster src/main.c src/recomp/startup.c)
    target_include_directories(coaster PRIVATE
        ${CMAKE_SOURCE_DIR}/include
        ${CMAKE_SOURCE_DIR}/RecompiledFuncs
    )
    target_link_libraries(coaster PRIVATE
        coaster_recomp
        coaster_hal
        coaster_platform
    )
    if(WIN32)
        target_link_libraries(coaster PRIVATE SDL2::SDL2main)
    endif()

    message(STATUS "RecompiledFuncs/ found - building 'coaster' executable")
else()
    message(STATUS "No recompiled sources - run: python tools/recomp.py COASTER_unpacked.exe RecompiledFuncs")
endif()
